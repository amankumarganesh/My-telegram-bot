<body>
  <h2>Tic-Tac-Toe</h2>
  <div id="status">Turn: Player X</div>
  <div class="grid" id="board"></div>
  <button onclick="resetGame()">Restart</button>
  <script>
    const bEl = document.getElementById('board');
    const sEl = document.getElementById('status');
    let b = Array(9).fill(""), p = "X", over = false;
    const wins = [[0,1,2],[3,4,5],[6,7,8],[0,3,6],[1,4,7],[2,5,8],[0,4,8],[2,4,6]];
    function render() {
      bEl.innerHTML = "";
      b.forEach((v, i) => {
        const c = document.createElement('div');
        c.className = 'cell';
        c.innerText = v;
        c.onclick = () => move(i);
        bEl.appendChild(c);
      });
    }
    function move(i) {
      if(b[i] !== "" || over) return;
      b[i] = p;
      if(wins.some(w => w.every(idx => b[idx] === p))) {
        sEl.innerText = "Player " + p + " Wins!";
        over = true;
      } else if(b.every(x => x !== "")) {
        sEl.innerText = "Draw!";
        over = true;
      } else {
        p = p === "X" ? "O" : "X";
        sEl.innerText = "Turn: Player " + p;
      }
      render();
    }
    function resetGame() {
      b = Array(9).fill("");
      p = "X";
      over = false;
      sEl.innerText = "Turn: Player X";
      render();
    }
    render();
  </script>
